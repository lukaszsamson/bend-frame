def rd(+tag: String, +path: String, +ts: List<&2, D.DType>) -> IO(Unit):
  do IO<Unit>:
    r : E.ResT(C.Frame) <- DF.read_csv_schema(path, DF.csv_opts(',', True{}, 100), ts)
    IO.print(R.show(tag, "eager", r))

def main() -> IO(Unit):
  do IO<Unit>:
    rd("B02-i count", "tests/regress/data/B02_h_crlf.csv", [D.TStr{}])
    rd("B02-i explicit", "tests/regress/data/B02_h_crlf.csv", [D.TStr{}, D.TI32{}])
    rd("B02-i bad-cell", "tests/regress/data/B02_h_crlf.csv", [D.TU32{}, D.TU32{}])
    # "" stays the empty string under an explicit str schema; a numeric "" is null (B02-a)
    rd("B02-i quoted-empty", "tests/regress/data/B02_a_numeric.csv", [D.TStr{}, D.TStr{}])
    rd("B02-i quoted-empty-u32", "tests/regress/data/B02_a_numeric.csv", [D.TU32{}, D.TStr{}])
