def chk(m: Maybe<&2, E.Err>) -> String:
  match m:
    case None{}:
      "none"
    case Some{e}:
      "some " ++ E.Err.show(e)

# 2900 equal keys on each side: 2900 * 2900 = 8410000 output rows, refused before allocating
def keys.go(n: Nat) -> List<&2, D.Scalar>:
  match n:
    case 0n:
      Nil{}
    case 1n+p:
      R.u(1) <> keys.go(p)

def side() -> E.ResT(C.Frame):
  R.frame([R.mk("k", keys.go(2900n), D.TU32{})])

def main() -> IO(Unit):
  do IO<Unit>:
    IO.print("-- B04-i guard")
    IO.print(U32.show(K.Kernel.max_rows()))
    IO.print(chk(K.Kernel.check_rows(8388608)))
    IO.print(chk(K.Kernel.check_rows(8388609)))
    IO.print("")
    IO.print(R.show("B04-i join", "eager", R.e_inner(side(), side(), ["k"], ["k"])))
    IO.print(R.show("B04-i concat-zero-col", "eager", DF.concat_vertical([C.Frame{8388608, Nil{}}, C.Frame{1, Nil{}}])))
    IO.print(R.show("B04-i concat-at-cap", "eager", DF.concat_vertical([C.Frame{8388607, Nil{}}, C.Frame{1, Nil{}}])))
